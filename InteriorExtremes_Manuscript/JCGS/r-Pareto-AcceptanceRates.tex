\title{Acceptance probabilities of rejection samplers for $r$-Pareto processes}
\date{\today}
\maketitle

\noindent This note collects (a) the acceptance probability of the sampler that turns $r_p$-Pareto draws into $r_0$-Pareto draws, which underlies Theorem~\ref{M-thm:acceptance} of the manuscript, and (b) the acceptance probability of the convex-set sampler of Theorem~\ref{M-thm:r_pareto_simulation}, which turns $r_0$-Pareto draws into $r$-Pareto draws for a convex risk functional $r$. Both are instances of one rejection step (Lemma~\ref{lem:generic}), and for $L_p$ norms their acceptance probabilities multiply to $D^{1/p-1}$ (Corollary~\ref{cor:duality}).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Setting}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Let $\Omega=\real^D_+\setminus\{\bm 0\}$ and let $\upkappa$ be an exponent measure on $\Omega$, homogeneous of order $-1$, i.e., $\upkappa(tB)=t^{-1}\upkappa(B)$ for $t>0$, with unit Pareto margins, $\upkappa(\{\bx\in\Omega:x_j>1\})=1$, $j=1,\dots,D$. A risk functional is a measurable $r:\Omega\to[0,\infty)$, homogeneous of order $1$, with $0<\upkappa(\calA_r)<\infty$, where $\calA_r=\{\bx\in\Omega:r(\bx)>1\}$. The $r$-Pareto random vector $\Zr{r}$ has distribution $\upkappa(\cdot\cap\calA_r)/\upkappa(\calA_r)$, and all $r$-Pareto vectors below share the same $\upkappa$. As in the manuscript, $r_0(\bx)=D^{-1}\|\bx\|_1$ is the spatial mean, $r_p(\bx)=(D^{-1}\sum_{j=1}^Dx_j^p)^{1/p}=D^{-1/p}\|\bx\|_p$ for $p\in(1,\infty)$ and $r_\infty(\bx)=\|\bx\|_\infty$, with the convention $1/\infty=0$. We write $\calA_0=\calA_{r_0}$ and $\calA_p=\calA_{r_p}$.

Write $\bx=s\bw$ with $s=\|\bx\|_1$ and $\bw\in\Delta=\{\bw\in\real^D_+:\|\bw\|_1=1\}$, so that $\upkappa(\text ds,\text d\bw)=s^{-2}\text ds\,H(\text d\bw)$, where $H$ is the angular measure of $\upkappa$.

\begin{enumerate}[(F1)]
\item Since $x_j>1$ if and only if $s>1/w_j$, unit Pareto margins are equivalent to
$$
1=\int_\Delta\int_{1/w_j}^\infty s^{-2}\text ds\,H(\text d\bw)=\int_\Delta w_j\,H(\text d\bw),\qquad j=1,\dots,D.
$$
Hence $H(\Delta)=\sum_{j=1}^D\int_\Delta w_j\,H(\text d\bw)=D$ and $\upkappa(\calA_0)=\int_\Delta\int_D^\infty s^{-2}\text ds\,H(\text d\bw)=H(\Delta)/D=1$.
\item For $t\geq1$ and Borel $B\subseteq\Delta$, $\Pr\{r_0(\Zr{r_0})>t,\ \Zr{r_0}/\|\Zr{r_0}\|_1\in B\}=(Dt)^{-1}H(B)=t^{-1}H(B)/D$. Thus $\Zr{r_0}=DR\bW$, where $R=r_0(\Zr{r_0})$ is unit Pareto, $\bW=\Zr{r_0}/\|\Zr{r_0}\|_1\sim H/D$, $R$ and $\bW$ are independent, and $\E\bW=D^{-1}\one$ by (F1).
\end{enumerate}

\begin{lemma}\label{lem:normconst}
For any risk functional $r$, $\upkappa(\calA_r)=\int_\Delta r(\bw)\,H(\text d\bw)=D\,\E\, r(\bW)$.
\end{lemma}

\begin{proof}
Since $r(s\bw)=s\,r(\bw)$, we have $s\bw\in\calA_r$ if and only if $s>1/r(\bw)$ (no $s$ if $r(\bw)=0$), so $\upkappa(\calA_r)=\int_\Delta\int_{1/r(\bw)}^\infty s^{-2}\text ds\,H(\text d\bw)=\int_\Delta r(\bw)\,H(\text d\bw)$. The second equality follows from (F2).
\end{proof}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{A generic rejection step}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{lemma}\label{lem:generic}
Let $r_a$ and $r_b$ be risk functionals and $t>0$ such that
\begin{equation}\label{eq:inclusion}
t\calA_{r_b}=\{\bx\in\Omega:r_b(\bx)>t\}\subseteq\calA_{r_a}.
\end{equation}
Then
\begin{enumerate}[(a)]
\item $t^{-1}\Zr{r_a}\ |\ r_b\big(\Zr{r_a}\big)>t\ \eqd\ \Zr{r_b}$;
\item $\Pr\big\{r_b\big(\Zr{r_a}\big)>t\big\}=t^{-1}\upkappa(\calA_{r_b})/\upkappa(\calA_{r_a})$.
\end{enumerate}
Moreover, \eqref{eq:inclusion} holds if and only if $r_b\leq t\,r_a$ on $\Omega$, i.e., if and only if $t\geq M_{ba}:=\sup\{r_b(\bx):\bx\in\Omega,\ r_a(\bx)\leq1\}$. Since (b) is decreasing in $t$, the largest acceptance probability is obtained with $t=M_{ba}$.
\end{lemma}

\begin{proof}
For Borel $B\subseteq\Omega$, $\{t^{-1}\Zr{r_a}\in B,\ r_b(\Zr{r_a})>t\}=\{\Zr{r_a}\in t(B\cap\calA_{r_b})\}$, and $t(B\cap\calA_{r_b})\subseteq t\calA_{r_b}\subseteq\calA_{r_a}$ by \eqref{eq:inclusion}. By homogeneity of $\upkappa$,
$$
\Pr\big\{t^{-1}\Zr{r_a}\in B,\ r_b\big(\Zr{r_a}\big)>t\big\}
=\frac{\upkappa\big(t(B\cap\calA_{r_b})\cap\calA_{r_a}\big)}{\upkappa(\calA_{r_a})}
=\frac{\upkappa\big(t(B\cap\calA_{r_b})\big)}{\upkappa(\calA_{r_a})}
=t^{-1}\frac{\upkappa(B\cap\calA_{r_b})}{\upkappa(\calA_{r_a})}.
$$
Taking $B=\Omega$ gives (b), and dividing by (b) gives $\upkappa(B\cap\calA_{r_b})/\upkappa(\calA_{r_b})=\Pr\{\Zr{r_b}\in B\}$, i.e., (a).
For the equivalence, \eqref{eq:inclusion} states that $r_a(\bx)\leq1$ implies $r_b(\bx)\leq t$. If $r_a(\bx)>0$, applying this to $\bx/r_a(\bx)$ gives $r_b(\bx)\leq t\,r_a(\bx)$; if $r_a(\bx)=0$, applying it to $\lambda\bx$, $\lambda\to\infty$, gives $r_b(\bx)=0$. The converse is immediate.
\end{proof}

\begin{remark}\label{rem:generic}
(i) If $\upkappa$ is homogeneous of order $-\alpha$, $\alpha>0$, the same proof gives (a), and (b) with $t^{-\alpha}$ in place of $t^{-1}$.
(ii) If $t<M_{ba}$, there exists $\bx$ with $r_a(\bx)\leq1$ and $r_b(\bx)>t$, and the accepted draws miss $t\calA_{r_b}\setminus\calA_{r_a}$, which is charged by some $\upkappa$ with unit Pareto margins (e.g., one whose angular measure has an atom at $\bx/\|\bx\|_1$). Hence $t\geq M_{ba}$ is needed for the sampler to be exact for every such $\upkappa$.
(iii) Theorem~\ref{M-thm:acceptance} is Lemma~\ref{lem:generic} with $(r_a,r_b,t)=(r_p,r_0,u)$; the convex-set sampler of Theorem~\ref{M-thm:r_pareto_simulation} has $(r_a,r_b,t)=(r_0,r,M)$ with $M=D\max_ir(\bm e_i)$; and \citet[Proposition~2]{Dombry.etal:2024} has $(r_a,r_b,t)=(r_0,r,M')$ for any $M'$ with $r\leq M'r_0$.
\end{remark}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{From $r_p$-Pareto to $r_0$-Pareto (Theorem~\ref{M-thm:acceptance})}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{proposition}\label{prop:rp_to_r0}
Let $p\in(1,\infty]$ and $u\geq1$.
\begin{enumerate}[(i)]
\item $u^{-1}\Zr{r_p}\ |\ r_0\big(\Zr{r_p}\big)>u\ \eqd\ \Zr{r_0}$.
\item $\rho_p(u):=\Pr\big\{r_0\big(\Zr{r_p}\big)>u\big\}=u^{-1}\upkappa(\calA_0)/\upkappa(\calA_p)=u^{-1}/\upkappa(\calA_p)$, which is decreasing in $u$. The choice $u=1$ maximises it, giving
$$
\rho_p:=\rho_p(1)=\frac{1}{\upkappa(\calA_p)}=\frac{D^{1/p-1}}{\E\|\bW\|_p}.
$$
\item $\upkappa(\calA_p)=\int_\Delta r_p(\bw)\,H(\text d\bw)=D^{1-1/p}\,\E\|\bW\|_p$ and $1\leq\upkappa(\calA_p)\leq D^{1-1/p}$, so that $\rho_p\in[D^{1/p-1},1]$. The value $\rho_p=1$ is attained under complete dependence, $H=D\,\delta_{D^{-1}\one}$, and $\rho_p=D^{1/p-1}$ under asymptotic independence, $H=\sum_{j=1}^D\delta_{\bm e_j}$.
\end{enumerate}
\end{proposition}

\begin{proof}
By the power mean (or Jensen's) inequality, $r_0\leq r_p$, with equality at $\bx=\one$, so $M_{ba}=1$ in Lemma~\ref{lem:generic} with $(r_a,r_b)=(r_p,r_0)$. Explicitly, for $u\geq1$, $r_0(\bx)>u$ implies $r_p(\bx)\geq r_0(\bx)>u\geq1$, i.e., $u\calA_0\subseteq\calA_p$. Lemma~\ref{lem:generic} then gives (i) and the first expression in (ii), and $\upkappa(\calA_0)=1$ by (F1). The last expression in (ii) follows from (iii).
For (iii), Lemma~\ref{lem:normconst} with $r=r_p$ gives the integral representation, and $r_p(\bw)=D^{-1/p}\|\bw\|_p$. For $\bw\in\Delta$, $D^{-1}=r_0(\bw)\leq r_p(\bw)\leq D^{-1/p}\|\bw\|_1=D^{-1/p}$, and integrating against $H$, with $H(\Delta)=D$, gives $1\leq\upkappa(\calA_p)\leq D^{1-1/p}$. Equality holds in the lower bound when $H$ is concentrated on $D^{-1}\one$, where $r_p(D^{-1}\one)=D^{-1}$, and in the upper bound when $H$ is concentrated on the vertices, where $r_p(\bm e_j)=D^{-1/p}$.
\end{proof}

\begin{remark}\label{rem:pinf}
For $p=\infty$, $\upkappa(\calA_\infty)=\upkappa\big([\bm 0,\one]^c\big)=V(\one)=\theta_D$, the $D$-variate extremal coefficient. Hence $\rho_\infty=1/\theta_D$: the proportion of exceedances of the maximum that are retained for the spectral likelihood is the reciprocal of the $D$-variate extremal coefficient.
\end{remark}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{From $r_0$-Pareto to $r$-Pareto with convex $r$ (Theorem~\ref{M-thm:r_pareto_simulation})}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

As in Theorem~\ref{M-thm:r_pareto_simulation}, let $r$ be convex and homogeneous of order $1$, with $r(\bm 0)=0$, and let $M=D\max_{i}r(\bm e_i)=\max_ir(D\bm e_i)>0$.

\begin{lemma}\label{lem:convex}
$r$ is subadditive, and for all $\bw\in\Delta$,
\begin{equation}\label{eq:convex_bound}
r(\bw)\leq\sum_{j=1}^D w_j\,r(\bm e_j)\leq\max_j r(\bm e_j)=\frac{M}{D}.
\end{equation}
Consequently, $r\leq Mr_0$ on $\Omega$, so \eqref{eq:inclusion} holds with $(r_a,r_b,t)=(r_0,r,M)$. Since the bound \eqref{eq:convex_bound} is attained at a vertex, $M_{ba}=M$.
\end{lemma}

\begin{proof}
Convexity and homogeneity give $r(\bx+\bm y)=2\,r\{(\bx+\bm y)/2\}\leq r(\bx)+r(\bm y)$. Hence, using $r(\bm 0)=0$ when $w_j=0$, $r(\bw)=r\big(\sum_jw_j\bm e_j\big)\leq\sum_jr(w_j\bm e_j)=\sum_jw_j\,r(\bm e_j)$. The second inequality in \eqref{eq:convex_bound} holds since $\bw\in\Delta$, and $r(\bx)=\|\bx\|_1\,r(\bx/\|\bx\|_1)\leq\|\bx\|_1M/D=Mr_0(\bx)$ for $\bx\in\Omega$.
\end{proof}

Lemma~\ref{lem:convex} is the analytic form of the convex-set argument in Section~E.1 of the Supplemental Material, namely $\calA_{r_0}^c=\mathrm{conv}\{\bm 0,D\bm e_1,\dots,D\bm e_D\}\subseteq\calA_{r/M}^c$.

\begin{proposition}\label{prop:r0_to_r}
Under the above conditions, with $\bW=\Zr{r_0}/\|\Zr{r_0}\|_1\sim H/D$:
\begin{enumerate}[(i)]
\item $\Zr{r_0}/M\ |\ r\big(\Zr{r_0}\big)>M\ \eqd\ \Zr{r}$;
\item the acceptance probability is
$$
\pi_r:=\Pr\big\{r\big(\Zr{r_0}\big)>M\big\}=\frac{\upkappa(\calA_r)}{M\,\upkappa(\calA_0)}=\frac1M\int_\Delta r(\bw)\,H(\text d\bw)=\frac{\E\,r(\bW)}{\max_jr(\bm e_j)};
$$
\item $\displaystyle \frac{r(\one)}{D\max_jr(\bm e_j)}\ \leq\ \pi_r\ \leq\ \frac{\sum_{j=1}^Dr(\bm e_j)}{D\max_jr(\bm e_j)}\ \leq\ 1$. The lower bound is attained under complete dependence and the upper bound under asymptotic independence, so both bounds are sharp over exponent measures with unit Pareto margins;
\item for $r=r_p$, $p\in(1,\infty]$, $M=D^{1-1/p}$ and
$$
\pi_p:=\Pr\big\{r_p\big(\Zr{r_0}\big)>D^{1-1/p}\big\}=\frac{\upkappa(\calA_p)}{D^{1-1/p}}=\E\|\bW\|_p\in[D^{1/p-1},1],
$$
with $\pi_p=D^{1/p-1}$ under complete dependence and $\pi_p=1$ under asymptotic independence; for $p=\infty$, $\pi_\infty=\theta_D/D$;
\item for $r(\bx)=\sum_jm_jx_j$, $m_j\geq0$, $\max_jm_j>0$, $M=D\max_jm_j$ and $\pi_r=\bar m/\max_jm_j$ with $\bar m=D^{-1}\sum_jm_j$, whatever the dependence structure.
\end{enumerate}
\end{proposition}

\begin{proof}
(i)--(ii) By Lemma~\ref{lem:convex}, \eqref{eq:inclusion} holds with $(r_a,r_b,t)=(r_0,r,M)$, and Lemma~\ref{lem:generic} gives (i) and the first expression in (ii). The others follow from $\upkappa(\calA_0)=1$ (F1) and Lemma~\ref{lem:normconst}. Alternatively, by (F2), $\Pr\{DR\,r(\bW)>M\mid\bW\}=\min\{1,D\,r(\bW)/M\}=D\,r(\bW)/M$, since $D\,r(\bW)\leq M$ by \eqref{eq:convex_bound}.

(iii) Since $r$ is convex and $\E\bW=D^{-1}\one$ (F2), Jensen's inequality gives $\E\,r(\bW)\geq r(D^{-1}\one)=r(\one)/D$. Taking expectations in \eqref{eq:convex_bound} gives $\E\,r(\bW)\leq\sum_jr(\bm e_j)\E W_j=D^{-1}\sum_jr(\bm e_j)\leq\max_jr(\bm e_j)$. Under complete dependence, $\bW=D^{-1}\one$ almost surely and Jensen's inequality is an equality; under asymptotic independence, $\bW$ is uniform on $\{\bm e_1,\dots,\bm e_D\}$ and the first inequality in \eqref{eq:convex_bound} is an equality almost surely.

(iv) $r_p(\bm e_j)=D^{-1/p}$ gives $M=D^{1-1/p}$, $r_p(\one)=1$, and $\upkappa(\calA_p)=D^{1-1/p}\E\|\bW\|_p$ by Proposition~\ref{prop:rp_to_r0}(iii); the bounds follow from (iii), and the case $p=\infty$ from Remark~\ref{rem:pinf}.

(v) $r(\bW)$ is linear in $\bW$, so $\E\,r(\bW)=\sum_jm_j\E W_j=\bar m$.
\end{proof}

\begin{remark}[Comparison with \citealp{Dombry.etal:2024}, and Table~\ref{M-tab:simu_exceedances_p_d}]
By Lemma~\ref{lem:generic} with $(r_a,r_b,t)=(r_0,r,M')$, the sampler of \citet[Proposition~2]{Dombry.etal:2024} has acceptance probability $D\,\E\,r(\bW)/M'$, so its ratio to $\pi_r$ is $M/M'$ whatever the dependence structure. For $r=r_p$ with $M'=D^{p-1-1/p}$, which is the constant $D^{p-2}$ of \citet{Dombry.etal:2024} for $\|\cdot\|_p\leq D^{p-2}\|\cdot\|_1$ rescaled to $r_p$ and $r_0$, the ratio is $D^{1-1/p}/D^{p-1-1/p}=D^{2-p}$, which matches Table~\ref{M-tab:simu_exceedances_p_d}: for $p=3$, $52/4=13.0$, $29/16=1.81$ and $18/64=0.28$ (tabulated $13$, $1.80$, $0.27$); for $p=5$, $D=4$, $49/64=0.77$ (tabulated $0.76$); and for $p=2$ the two samplers coincide. For $(p,D)=(5,16)$ and $(10,4)$ the ratio gives $26/16^3\approx0.006$ and $47/4^8\approx0.0007$, so these entries are below $0.01\%$.
\end{remark}

\section{Complementarity of the two samplers}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{corollary}\label{cor:duality}
For $p\in(1,\infty]$ and every exponent measure with unit Pareto margins,
$$
\rho_p\,\pi_p=D^{1/p-1},
$$
where $\rho_p$ is the acceptance probability of the $r_p\to r_0$ sampler with $u=1$ (Proposition~\ref{prop:rp_to_r0}) and $\pi_p$ is that of the $r_0\to r_p$ sampler (Proposition~\ref{prop:r0_to_r}(iv)). Complete dependence gives $(\rho_p,\pi_p)=(1,D^{1/p-1})$, and asymptotic independence gives $(\rho_p,\pi_p)=(D^{1/p-1},1)$.
\end{corollary}

\begin{proof}
$\rho_p=1/\upkappa(\calA_p)$ and $\pi_p=\upkappa(\calA_p)/D^{1-1/p}$.
\end{proof}

\begin{remark}\label{rem:practice}
(i) The empirical acceptance rate of the simulation step is a Monte Carlo estimate of $\pi_p=\E\|\bW\|_p$, so the proportion of $r_p$ exceedances retained by the spectral likelihood under the simulated model follows at no extra cost as $\rho_p=D^{1/p-1}/\pi_p$. For the Brown--Resnick model of Table~\ref{M-tab:simu_exceedances_p_d} with $D=100$, $\pi_3\approx0.16$ gives $\rho_3\approx100^{-2/3}/0.16\approx0.29$, and $\pi_{10}\approx0.13$ gives $\rho_{10}\approx100^{-0.9}/0.13\approx0.12$.

(ii) More generally, for a convex $r$ with $m_r:=\inf_{\bw\in\Delta}r(\bw)>0$, the sharpest constant in $r_0\leq u\,r$ is $M_{ba}=1/(Dm_r)$ (Lemma~\ref{lem:generic} with $(r_a,r_b)=(r,r_0)$). The $r\to r_0$ sampler with $u=1/(Dm_r)$ then has acceptance probability $\rho_r=m_r/\E\,r(\bW)$ by Lemma~\ref{lem:generic}(b) and Lemma~\ref{lem:normconst}, and $\rho_r\pi_r=m_r/\max_jr(\bm e_j)$. For $r=r_p$, $m_r=D^{-1}$ and $\max_jr(\bm e_j)=D^{-1/p}$, which recovers Corollary~\ref{cor:duality}.
\end{remark}
